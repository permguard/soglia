			     10.201.0.5 . 8080 }

	chain prerouting {
		type filter hook prerouting priority raw; policy accept;
		iifname "sgh-*" meta nfproto ipv6 drop

	chain forward {
		type filter hook forward priority filter - 10; policy accept;
		iifname "sgh-*" drop
		oifname "sgh-*" drop
	}

	chain output {
		type filter hook output priority filter - 10; policy accept;
		oifname "sgh-*" meta nfproto ipv6 drop
		oifname "sgh-*" ct state invalid drop
		oifname "sgh-*" ct state new meta l4proto != tcp drop
		oifname "sgh-*" ct state new meta skuid != 990 drop
		oifname "sgh-*" ct state new ip daddr . tcp dport != @exec_ingress drop
	}
